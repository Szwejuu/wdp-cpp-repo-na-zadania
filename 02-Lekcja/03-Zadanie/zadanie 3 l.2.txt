#include <iostream>

using namespace std;

int main() {
    double a, b;
    cout << "Podaj dwie liczby rzeczywiste (druga różna od zera): ";
    cin >> a >> b;

    cout << "Suma: " << a + b << endl;
    cout << "Różnica: " << a - b << endl;
    cout << "Iloczyn: " << a * b << endl;
    cout << "Iloraz: " << a / b << endl;

    return 0;
}
